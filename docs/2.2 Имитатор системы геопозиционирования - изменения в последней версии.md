# Обновлённый код: панель «проблемных» ячеек справа

Что добавляется:

1. **Детектор «проблемных» ячеек.** Для каждой ячейки в видимой области сетки определяем её состояние:
   - `ok` — QR распознан (текст в кэше, ячейка видна),
   - `covered` — внутри ячейки найден **другой** QR-код (координаты не совпадают с координатами ячейки → кто-то перекрыт),
   - `empty` — QR не найден и в кэше текста нет.
2. **Панель справа** — полупрозрачная колонка поверх кадра, где обновляется список:
   - `ЗАКРЫТО: A3 (сверху B2)` — в ячейке `A3` лежит QR от `B2`,
   - `ПУСТО: D7` — QR не обнаружен и в кэше нет.

3. **Алгоритм определения перекрытия**: для каждого распознанного QR мы знаем его квад и «родные» координаты (из текста). Если центр этого QR попадает внутрь геометрической ячейки с **другими** координатами — значит, ячейка закрыта.

## Код

```python
import cv2
import numpy as np
import re

# ------------------- Параметры -------------------
COLS = 20
ROWS = 20
LETTERS = "ABCDEFGHIJKLMNOPQRST"

CAM_INDEX = 0
FRAME_W = 1280
FRAME_H = 720

ALPHA = 0.7
CACHE_TTL = 0  # 0 = помнить бесконечно

PANEL_W = 340          # ширина панели справа
FONT = cv2.FONT_HERSHEY_SIMPLEX

# ------------------- Утилиты -------------------
def parse_cell(name: str):
    name = name.strip().upper()
    m = re.fullmatch(r"([A-T])(\d{1,2})", name)
    if not m:
        return None
    row = LETTERS.index(m.group(1))
    col = int(m.group(2)) - 1
    if not (0 <= row < ROWS and 0 <= col < COLS):
        return None
    return row, col

def cell_name(row, col):
    return f"{LETTERS[row]}{col + 1}"

def point_in_poly(pt, poly):
    """poly: np.ndarray (N,2). Возвращает True, если pt внутри."""
    return cv2.pointPolygonTest(poly.astype(np.float32), pt, False) >= 0


detector = cv2.QRCodeDetector()
try:
    detector.setUseAlignmentMarkers(True)
except Exception:
    pass

cap = cv2.VideoCapture(CAM_INDEX, cv2.CAP_DSHOW)
cap.set(cv2.CAP_PROP_FRAME_WIDTH, FRAME_W)
cap.set(cv2.CAP_PROP_FRAME_HEIGHT, FRAME_H)

if not cap.isOpened():
    raise RuntimeError("Не удалось открыть камеру")

# -------- Кэши --------
cell_text = {}
cell_quad = {}
cell_age  = {}

smooth_H = None
frame_idx = 0

print("ESC/Q — выход, R — сброс кэша.")

while True:
    ok, frame = cap.read()
    if not ok:
        break
    frame_idx += 1

    ok_qr, decoded_info, points, _ = detector.detectAndDecodeMulti(frame)

    # -------- Все распознанные QR (даже если не в диапазоне) --------
    # Список: (row, col, quad, text)
    detected = []
    visible = set()

    if ok_qr and points is not None and decoded_info is not None:
        for text, quad in zip(decoded_info, points):
            if not text:
                continue
            parsed = parse_cell(text)
            if parsed is None:
                continue
            row, col = parsed
            quad = quad.astype(np.float32)

            cell_text[(row, col)] = text
            cell_quad[(row, col)] = quad
            cell_age[(row, col)] = 0
            visible.add((row, col))
            detected.append((row, col, quad, text))

            poly = quad.astype(np.int32).reshape(-1, 1, 2)
            cv2.polylines(frame, [poly], True, (0, 255, 0), 2)
            cv2.putText(frame, text,
                        (int(quad[:, 0].min()), int(quad[:, 1].min()) - 8),
                        FONT, 0.8, (0, 255, 0), 2)

    # старение кэша
    for key in list(cell_age.keys()):
        if key not in visible:
            cell_age[key] += 1
            if CACHE_TTL > 0 and cell_age[key] > CACHE_TTL:
                cell_text.pop(key, None)
                cell_quad.pop(key, None)
                cell_age.pop(key, None)

    # -------- Гомография --------
    H = None
    if len(visible) >= 4:
        rows = [r for (r, c) in visible]
        cols = [c for (r, c) in visible]
        min_r, max_r = min(rows), max(rows)
        min_c, max_c = min(cols), max(cols)

        def center_of(r, c):
            q = cell_quad.get((r, c))
            if q is None:
                return None
            return (float(q[:, 0].mean()), float(q[:, 1].mean()))

        tl = center_of(min_r, min_c)
        tr = center_of(min_r, max_c)
        br = center_of(max_r, max_c)
        bl = center_of(max_r, min_c)

        if tl and tr and br and bl:
            src = np.float32([tl, tr, br, bl])
            dst = np.float32([
                [min_c,     min_r],
                [max_c + 1, min_r],
                [max_c + 1, max_r + 1],
                [min_c,     max_r + 1],
            ])
            H = cv2.getPerspectiveTransform(dst, src)

    if H is not None:
        smooth_H = H if smooth_H is None else ALPHA * smooth_H + (1 - ALPHA) * H

    # -------- Анализ ячеек --------
    # statuses: dict (row, col) -> ("ok" | "covered" | "empty", extra)
    statuses = {}

    if smooth_H is not None:
        def w2s(col_f, row_f):
            p = np.array([col_f, row_f, 1.0], dtype=np.float32)
            q = smooth_H @ p
            q /= q[2]
            return int(q[0]), int(q[1])

        known = visible | set(cell_text.keys())
        if known:
            r0 = min(r for (r, c) in known)
            r1 = max(r for (r, c) in known)
            c0 = min(c for (r, c) in known)
            c1 = max(c for (r, c) in known)
        else:
            r0, r1, c0, c1 = 0, 0, 0, 0

        # --- сетка ---
        for r in range(r0, r1 + 2):
            cv2.line(frame, w2s(c0, r), w2s(c1 + 1, r), (0, 255, 255), 2)
        for c in range(c0, c1 + 2):
            cv2.line(frame, w2s(c, r0), w2s(c, r1 + 1), (0, 255, 255), 2)

        # --- для каждой ячейки вычислим квад и определим статус ---
        cell_corners = {}
        for r in range(r0, r1 + 1):
            for c in range(c0, c1 + 1):
                cell_corners[(r, c)] = np.float32([
                    w2s(c,     r),
                    w2s(c + 1, r),
                    w2s(c + 1, r + 1),
                    w2s(c,     r + 1),
                ])

        # Найдём все распознанные QR, чей центр лежит внутри данной ячейки,
        # но чьи «родные» координаты — другие. Это признак перекрытия.
        covered_by = {}  # (row, col) -> имя чужого QR, который лежит сверху
        for (qr_row, qr_col, quad, text) in detected:
            cx = float(quad[:, 0].mean())
            cy = float(quad[:, 1].mean())
            for key, poly in cell_corners.items():
                if key == (qr_row, qr_col):
                    continue
                if point_in_poly((cx, cy), poly):
                    # чужой QR лежит в этой ячейке
                    covered_by.setdefault(key, set()).add(text)

        # --- статусы и отрисовка ---
        for key, poly in cell_corners.items():
            r, c = key
            text = cell_text.get(key)

            if key in covered_by:
                # перекрытие — приоритетнее
                statuses[key] = ("covered", sorted(covered_by[key]))
                cv2.polylines(frame, [poly.astype(np.int32).reshape(-1,1,2)],
                              True, (0, 128, 255), 3)  # оранжевый контур
            elif text is not None:
                statuses[key] = ("ok", text)
                cx = int(poly[:, 0].mean())
                cy = int(poly[:, 1].mean())
                cv2.putText(frame, text, (cx - 24, cy + 10),
                            FONT, 1.1, (0, 0, 0), 5, cv2.LINE_AA)
                cv2.putText(frame, text, (cx - 24, cy + 10),
                            FONT, 1.1, (255, 255, 255), 2, cv2.LINE_AA)
            else:
                statuses[key] = ("empty", None)
                cv2.line(frame, tuple(poly[0].astype(int)),
                         tuple(poly[2].astype(int)), (0, 0, 255), 2, cv2.LINE_AA)
                cv2.line(frame, tuple(poly[1].astype(int)),
                         tuple(poly[3].astype(int)), (0, 0, 255), 2, cv2.LINE_AA)

    # -------- Панель справа --------
    h, w = frame.shape[:2]
    canvas = np.zeros((h, w + PANEL_W, 3), dtype=np.uint8)
    canvas[:, :w] = frame
    canvas[:, w:] = (30, 30, 30)
    cv2.line(canvas, (w, 0), (w, h), (100, 100, 100), 2)

    px = w + 15
    py = 35

    cv2.putText(canvas, "ПРОБЛЕМНЫЕ ЯЧЕЙКИ", (px, py),
                FONT, 0.7, (255, 255, 255), 2)
    py += 30

    # Собираем списки
    covered_list = []
    empty_list = []
    for key, (st, info) in statuses.items():
        if st == "covered":
            covered_list.append((key, info))
        elif st == "empty":
            empty_list.append((key, None))

    covered_list.sort(key=lambda kv: (kv[0][0], kv[0][1]))
    empty_list.sort(key=lambda kv: (kv[0][0], kv[0][1]))

    # --- секция "ЗАКРЫТЫ" ---
    cv2.putText(canvas, f"ЗАКРЫТЫ ({len(covered_list)}):", (px, py),
                FONT, 0.6, (0, 165, 255), 2)
    py += 24

    if not covered_list:
        cv2.putText(canvas, "  —", (px, py), FONT, 0.55, (150, 150, 150), 1)
        py += 22
    else:
        for (r, c), overlays in covered_list:
            name = cell_name(r, c)
            label = f"{name} <- {','.join(overlays)}"
            if py > h - 40:
                cv2.putText(canvas, "...", (px, py), FONT, 0.55,
                            (200, 200, 200), 1)
                break
            cv2.putText(canvas, label, (px, py), FONT, 0.55,
                        (255, 200, 100), 1, cv2.LINE_AA)
            py += 22

    py += 15

    # --- секция "ПУСТО" ---
    cv2.putText(canvas, f"ПУСТО ({len(empty_list)}):", (px, py),
                FONT, 0.6, (0, 0, 255), 2)
    py += 24

    if not empty_list:
        cv2.putText(canvas, "  —", (px, py), FONT, 0.55, (150, 150, 150), 1)
        py += 22
    else:
        # выводим компактно: несколько имён в строке через запятую
        line = "  "
        for (r, c), _ in empty_list:
            name = cell_name(r, c) + ", "
            if len(line) + len(name) > 38:
                cv2.putText(canvas, line.rstrip(", "), (px, py), FONT, 0.55,
                            (150, 150, 255), 1, cv2.LINE_AA)
                py += 22
                line = "  "
                if py > h - 25:
                    break
            line += name
        if line.strip() and py <= h - 25:
            cv2.putText(canvas, line.rstrip(", "), (px, py), FONT, 0.55,
                        (150, 150, 255), 1, cv2.LINE_AA)

    # --- HUD слева сверху ---
    cv2.putText(canvas, f"Visible: {len(visible)}  Cached: {len(cell_text)}",
                (20, 30), FONT, 0.7, (255, 255, 255), 2)
    cv2.putText(canvas, "ESC/Q - exit, R - reset cache",
                (20, 60), FONT, 0.55, (200, 200, 200), 1)

    cv2.imshow("QR grid", canvas)
    key = cv2.waitKey(1) & 0xFF
    if key in (27, ord('q'), ord('Q')):
        break
    if key in (ord('r'), ord('R')):
        cell_text.clear()
        cell_quad.clear()
        cell_age.clear()
        smooth_H = None

cap.release()
cv2.destroyAllWindows()
```

## Логика определения «закрытых»

1. Детектор возвращает все видимые QR и их квады.
2. Из текста берём **истинные** координаты этого QR (`A1`, `B7`, …).
3. По гомографии восстанавливаем геометрический квад каждой ячейки видимой области.
4. Для каждого распознанного QR проверяем: в какой геометрической ячейке лежит его центр?
   - если это **та же** ячейка — всё хорошо,
   - если **другая** — значит, чужая ячейка закрыта этим QR.
5. `covered_by[(r,c)]` накапливает список QR-ов, которые лежат в ячейке `(r,c)`, но «родом» из других мест.

В панели выводится `A3 <- B2` — то есть «в A3 лежит QR от B2».

## Что рисуется на экране

| Состояние | Цвет | Что видно |
|---|---|---|
| `ok` — QR распознан | белый текст на чёрной подложке | имя ячейки в центре |
| `covered` — ячейка закрыта другим QR | оранжевый контур ячейки | в панели строка `имя <- чужие` |
| `empty` — кода нет | красный крест по диагоналям | в панели строка `ПУСТО` |

## Тонкости

- Если QR-код частично вылезает за свою ячейку и его центр оказался в соседней, может возникнуть ложное «перекрытие». Можно уменьшить зону проверки — например, брать не центр, а точку `0.5·(угол_QR + центр_ячейки)`, или проверять, что **весь** QR внутри ячейки (`pointPolygonTest(..., measureDist=True) > -0.2 * размер`).
- `CACHE_TTL = 0` — кэш не стареет, `ПУСТО` показывается только для тех ячеек, которые **никогда** не были распознаны за сессию. Если хотите периодически чистить — задайте TTL, например 300.
- Панель — отдельный массив `canvas`, чтобы не портить `frame`. Можно её сузить/расширить через `PANEL_W`.
